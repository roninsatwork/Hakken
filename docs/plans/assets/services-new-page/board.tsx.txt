type ChannelFigures = { v: number; g: number; e: number; vb: number; gb: number; eb: number; m: [number, number][] };

const add = (pick: (service: Service) => number, list: readonly Service[] = SERVICES) => list.reduce((sum, service) => sum + pick(service), 0);
const EVENT_KEYS = ["contact", "phone_number_click", "email_link_click", "live_chat_event"];
const touched = (s: Service, event = "any") => (event === "any" ? EVENT_KEYS.reduce((sum, key) => sum + (s.eventVisits[key] ?? 0), 0) : s.eventVisits[event] ?? 0);
const touchedBefore = (s: Service, event = "any") => (event === "any" ? EVENT_KEYS.reduce((sum, key) => sum + (s.eventVisitsBefore[key] ?? 0), 0) : s.eventVisitsBefore[event] ?? 0);
const T = {

const isEvery = (ticked: readonly string[]) => ticked.length === CHANNELS.length;
/** A service's figures for the channels ticked. Every channel is the page's own. A narrower pick can only count visits that started on the service. */
function through(s: Service, ticked: readonly string[]) {
  if (isEvery(ticked)) return { visits: s.visits, before: s.visitsBefore, engaged: s.engaged, engagedBefore: s.engagedBefore, events: s.events, eventsBefore: s.eventsBefore };
  const out = { visits: 0, before: 0, engaged: 0, engagedBefore: 0, events: 0, eventsBefore: 0 };
  for (const channel of ticked) {
    const c = s.channels[channel];
    if (!c) continue;
    out.visits += c.v; out.before += c.vb; out.engaged += c.g; out.engagedBefore += c.gb; out.events += c.e; out.eventsBefore += c.eb;
  }
  return out;
}

/** The kit's table on the whole list: its search, Show filter, sorting and pages all work, as on a built screen. */
function Table({ columns, opening, firsts, actions, filterBy, filters, sortBy }: {
  columns: Parameters<typeof ListTable<Service>>[0]["columns"];
  opening: string;
  firsts: Record<string, "asc" | "desc">;
  actions?: ReactNode;
  filterBy?: (s: Service) => boolean;
  filters?: ReactNode;
  sortBy?: Record<string, (s: Service) => number | string>;
}) {
  const order = useSiteSort(firsts, opening);
  const [search, setSearch, settled] = useSiteSearch();
  const [show, setShow] = useSiteParam<string>("show", "all", ["all", "events", "none"]);
  const term = settled.toLowerCase();
  const read = { ...SORT, ...(sortBy ?? {}) }[order.key] ?? SORT.visits;
  const sign = order.direction === "asc" ? 1 : -1;
  const rows = SERVICES
    .filter((s) => (!term || s.name.toLowerCase().includes(term) || s.key.includes(term)) && (filterBy ? filterBy(s) : show === "all" || (show === "events") === (s.events > 0)))
    .sort((a, b) => {
      const left = read(a), right = read(b);
      const moved = typeof left === "string" ? left.localeCompare(String(right)) : Number(left) - Number(right);
      return sign * moved || b.visits - a.visits;
    });
  const paged = useSitePager(rows);
  return (
    <ListTable
      rows={paged.pageRows}
      rowKey={(row) => row.key}
      noun={"services" as never}
      summary={<span className="text-[12px] text-secondary">Last 12 months · every device</span>}
      actions={actions ?? <DownloadButton label="Download CSV" busyLabel="Downloading…" onClick={noop} />}
      search={{ value: search, onChange: setSearch, placeholder: "Find a service…" }}
      filters={<div className="flex flex-wrap items-center gap-3">{filters ?? (
        <Select aria-label="Show" chip={{ label: "Show", choice: show === "all" ? null : SHOW_WORDS[show] }} value={show} onChange={setShow}>
          {Object.entries(SHOW_WORDS).map(([value, label]) => <option key={value} value={value}>{label}</option>)}
        </Select>
      )}</div>}
      sort={order.tableSort}
      onRowClick={noop}
      empty={{ icon: <BriefcaseBusiness className="h-8 w-8 text-muted/30" />, label: settled ? "Nothing matches that search." : "No services in these dates." }}
      footer={paged.footer}
      columns={columns}
    />
  );
}

function WayOneChart() {
  const [view, setView] = useState("most");
  const [ticked, setTicked] = useState(CHANNELS);
  const [many, setMany] = useState("10");
  const shown = VIEWS.find((one) => one.value === view)!;
  const figures = SERVICES.map((s) => ({ s, f: through(s, ticked) }));
  const limit = many === "all" ? SERVICES.length : Number(many);
  const label = (row: (typeof figures)[number]) => row.s.name;
  let rows: Array<Record<string, unknown> & { label: string }> = [];
  if (view === "most") rows = figures.filter((r) => r.f.events > 0).sort((a, b) => b.f.events - a.f.events || b.f.visits - a.f.visits).slice(0, limit).map((r) => ({ label: label(r), events: r.f.events }));
  if (view === "quiet") rows = figures.filter((r) => r.f.events === 0 && r.f.visits > 0).sort((a, b) => b.f.visits - a.f.visits).slice(0, limit).map((r) => ({ label: label(r), visits: r.f.visits }));
  if (view === "visits") rows = figures.filter((r) => r.f.visits > 0).sort((a, b) => b.f.visits - a.f.visits).slice(0, limit).map((r) => ({ label: label(r), visits: r.f.visits }));

  if (view === "more") rows = figures.filter((r) => r.f.events > r.f.eventsBefore).sort((a, b) => (b.f.events - b.f.eventsBefore) - (a.f.events - a.f.eventsBefore) || b.f.visits - a.f.visits).slice(0, limit).map((r) => ({ label: label(r), before: r.f.eventsBefore, events: r.f.events }));
  if (view === "fewer") rows = figures.filter((r) => r.f.events < r.f.eventsBefore).sort((a, b) => (a.f.events - a.f.eventsBefore) - (b.f.events - b.f.eventsBefore) || b.f.visits - a.f.visits).slice(0, limit).map((r) => ({ label: label(r), before: r.f.eventsBefore, events: r.f.events }));
  const series = view === "quiet"

const LINE_COLOURS = [CHART_SERIES_ORANGE, CHART_SERIES_BLUE, CHART_SERIES_TEAL, CHART_SERIES_VIOLET, CHART_SERIES_EMERALD, CHART_SERIES_AMBER, CHART_SERIES_ROSE, CHART_SERIES_SLATE];
const colourAt = (at: number) => LINE_COLOURS[at % LINE_COLOURS.length];
const monthEnd = (month: string) => {
  const [year, m] = month.split("-").map(Number);
  return new Date(Date.UTC(year, m, 0)).toISOString().slice(0, 10);
};
type MonthTotals = { v: number; w: number; e: number };
const SHOW_LABELS: Record<string, { label: string; figure: string; word: string }> = {
  visits: { label: "Visits", figure: "Visits", word: "Visits" },
  events: { label: "Events", figure: "Events", word: "Events" },
  rate: { label: "Events for every 100 visits", figure: "Per 100 visits", word: "Events for every 100 visits" },
  views: { label: "Page views", figure: "Page views", word: "Page views" },
};
function monthOf(list: readonly Service[], ticked: readonly string[], at: number): MonthTotals {
  const total: MonthTotals = { v: 0, w: 0, e: 0 };
  for (const s of list) {
    if (isEvery(ticked)) { total.v += s.months[at][0]; total.w += s.months[at][1]; total.e += s.months[at][2]; continue; }
    for (const channel of ticked) { const c = s.channels[channel]; if (c) { total.v += c.m[at][0]; total.e += c.m[at][1]; } }
  }
  return total;
}
function siteMonth(ticked: readonly string[], at: number): MonthTotals {
  if (isEvery(ticked)) return { v: DATA.site.months[at][0], w: DATA.site.months[at][1], e: DATA.site.months[at][2] };
  const total: MonthTotals = { v: 0, w: 0, e: 0 };
  for (const channel of ticked) { const m = (DATA.site.channelMonths as Record<string, number[][]>)[channel]; if (m) { total.v += m[at][0]; total.e += m[at][1]; } }
  return total;
}
const reading = (show: string, m: MonthTotals): number | null => (show === "visits" ? m.v : show === "views" ? m.w : show === "events" ? m.e : m.v > 0 ? Math.round((1000 * m.e) / m.v) / 10 : null);
const pointsOf = (show: string, at: readonly number[], read: (index: number) => MonthTotals) => at.flatMap((index) => {
  const value = reading(show, read(index));
  return value === null ? [] : [{ day: `${MONTHS[index]}-01`, lastDay: monthEnd(MONTHS[index]), value }];
});

/** Whole numbers that add up to the total: the biggest leftovers get the spare ones. */
function outOf(values: readonly number[], cells: number) {
  const whole = values.reduce((total, value) => total + value, 0);
  if (whole === 0) return values.map(() => 0);
  const exact = values.map((value) => (cells * value) / whole);
  const counts = exact.map(Math.floor);
  let spare = cells - counts.reduce((total, value) => total + value, 0);
  for (const { at } of exact.map((value, index) => ({ at: index, left: value - Math.floor(value) })).sort((a, b) => b.left - a.left)) { if (spare <= 0) break; counts[at] += 1; spare -= 1; }
  return counts;
}

const countryName = (code: string) => {
  if (code === "(other)") return "Others";
  try { return new Intl.DisplayNames(["en-GB"], { type: "region" }).of(code) ?? code; } catch { return code; }
};
const DEVICE_NAMES: Record<string, string> = { desktop: "Desktop", mobile: "Mobile", tablet: "Tablet" };
const COUNTRY_KEYS = [...(DATA.countries as string[]).slice(0, 6), "(other)"];
type Square = { visits: number; times: number; visitsBefore: number; timesBefore: number };
function squareFor(s: Service, across: string, key: string): Square {
  if (across === "channel") { const c = s.channels[key]; return { visits: c?.v ?? 0, times: c?.e ?? 0, visitsBefore: c?.vb ?? 0, timesBefore: c?.eb ?? 0 }; }
  if (across === "device") { const d = (s.device as Record<string, { v: number; e: number; vb: number; eb: number }>)[key]; return { visits: d?.v ?? 0, times: d?.e ?? 0, visitsBefore: d?.vb ?? 0, timesBefore: d?.eb ?? 0 }; }
  const c = (s.countries as unknown as Record<string, { w: number; wb: number }>)[key];
  return { visits: c?.w ?? 0, times: 0, visitsBefore: c?.wb ?? 0, timesBefore: 0 };
}

          { kind: "select", key: "across", label: "Count", value: acrossKey, options: Object.entries(ACROSS).map(([value, one]) => ({ value, label: one.label })), onChange: setAcross },
          { kind: "select", key: "up", label: "Show", value: upKey, options: Object.entries(DOING).map(([value, one]) => ({ value, label: one.show })), onChange: setUp },
          ...(canChannel ? [channelsControl(ticked, setTicked)] : []),
          { kind: "select", key: "busy", label: acrossKey === "views" ? "Page views from" : "Visits from", value: busyPick, options: [{ value: "0", label: `${fmt(lines[0])}, half your middle service` }, { value: "1", label: `${fmt(lines[1])}, your middle service` }, { value: "2", label: `${fmt(lines[2])}, twice your middle service` }], onChange: setBusyPick },
          { kind: "select", key: "working", label: doing.from, value: working, options: [{ value: "1", label: `${up.site.toFixed(1)}, your whole website` }, { value: "0.5", label: `${(up.site / 2).toFixed(1)}, half your website` }, { value: "2", label: `${(up.site * 2).toFixed(1)}, twice your website` }], onChange: setWorking },
        ]}
        message={channels.length === 0 ? "Tick a channel in the Channels menu to draw it here." : null}
      >
        <DotGroups
          points={shownServices.map((s) => ({ x: xOf(s), y: yOf(s) > 0 ? Math.round(yOf(s) * 10) / 10 : 0.5, label: s.name, named: Boolean(SAY[s.key]) }))}
          split={split}
          names={names}
          xLabel={across.label}
          yLabel={doing.axis}
          readout={{ x: across.label, y: "For every 100 visits" }}
          height={420}
        />
      </MapCard>
      <MapCard
        title="All your services"
        hint={`Bigger boxes had more ${BOX_WORDS[mapCount]}. Orange boxes got more ${BOX_WORDS[mapColour]} than in the 12 months before, and grey boxes got fewer.`}
        exportName="services-map"
        csv={csv}
        host={HOST}
        controls={[
          { kind: "select", key: "count", label: "Count", value: mapCount, options: Object.entries(COUNT_READ).map(([value, one]) => ({ value, label: one.label })), onChange: setMapCount },
          { kind: "select", key: "colour", label: "Colour shows", value: mapColour, options: Object.entries(COUNT_READ).map(([value, one]) => ({ value, label: one.label })), onChange: setMapColour },
          { kind: "select", key: "group", label: "Show", value: group, options: groupOptions, onChange: setGroup },
        ]}
        message={mapBoxes.length === 0 ? "No services to show for these dates." : null}
      >
        <PageMap boxes={mapBoxes} />
      </MapCard>
      <Table
        opening="visits"
        firsts={{ name: "asc", visits: "desc", visitsUp: "desc", events: "desc" }}
        filterBy={inGroup}
        filters={(
          <Select aria-label="Show" chip={{ label: "Show", choice: group ? names[group as DotGroupKey] : null }} value={group} onChange={setGroup}>
            {groupOptions.map((one) => <option key={one.value} value={one.value}>{one.label}</option>)}
          </Select>
        )}
        columns={[
          { key: "name", header: "Service", sortable: true, className: CUT_COLUMN.first, cell: (s: Service) => <ServiceCell service={s} /> },
          { key: "visits", header: "Visits", align: "right", sortable: true, cell: (s: Service) => <FigureCell value={s.visits} /> },
          { key: "visitsUp", header: "Up or down", align: "right", sortable: true, cell: (s: Service) => <UpOrDownCell now={s.visits} before={s.visitsBefore} /> },
          { key: "events", header: "Events", align: "right", sortable: true, cell: (s: Service) => <EventsCell service={s} /> },
        ]}
      />
    </Page>
  );
}
