// Builds the boards: today's page from its copy, and each way rendered in a real browser inside the same frame.
// node build.mjs <kit url on the canvas>
import { createRequire } from "node:module";
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
import path from "node:path";
import { pathToFileURL } from "node:url";

const REPO = "/Users/ants/Projects/Kakken";
const require = createRequire(`${REPO}/package.json`);
const { build } = require("esbuild");
const { chromium } = require("playwright");
const { board } = await import(pathToFileURL(`${REPO}/scripts/drawing-kit/from-app.mjs`).href);

const HERE = path.dirname(new URL(import.meta.url).pathname);
const OUT = path.join(HERE, "canvas/project");
mkdirSync(OUT, { recursive: true });
const KIT = process.argv[2] ?? "kit.css";
const STUBS = `${REPO}/scripts/drawing-kit/stubs`;
const STUBBED = {
  "next/link": "next-link.tsx", "next/navigation": "next-navigation.tsx", "next/image": "next-image.tsx", "next-intl": "next-intl.tsx",
  "framer-motion": "framer-motion.tsx", "convex/react": "convex-react.tsx", "@convex-dev/auth/react": "convex-auth-react.tsx",
  "next-themes": "next-themes.tsx", "@/src/context/SystemSettingsContext": "system-settings.tsx",
};

// 1. The ways' bundle, for a browser.
await build({
  entryPoints: [path.join(HERE, "board/main.tsx")],
  outfile: path.join(HERE, "board.js"),
  bundle: true,
  platform: "browser",
  format: "iife",
  jsx: "automatic",
  logLevel: "error",
  tsconfig: `${REPO}/tsconfig.json`,
  nodePaths: [`${REPO}/node_modules`],
  define: { "process.env.NODE_ENV": '"production"' },
  loader: { ".json": "json" },
  plugins: [{
    name: "drawing-kit-stubs",
    setup(builder) {
      const pattern = new RegExp(`^(${Object.keys(STUBBED).map((name) => name.replace(/[/.@-]/g, (c) => `\\${c}`)).join("|")})$`);
      builder.onResolve({ filter: pattern }, (args) => ({ path: path.join(STUBS, STUBBED[args.path]) }));
      builder.onResolve({ filter: /^@\// }, (args) => {
        const base = path.join(REPO, args.path.slice(2));
        for (const end of ["", ".tsx", ".ts", "/index.tsx", "/index.ts", ".json"]) {
          try { readFileSync(base + end); return { path: base + end }; } catch { /* next */ }
        }
        return undefined;
      });
    },
  }],
});

// 2. Today's board, from the copy. Its page area is what each way replaces.
const copy = readFileSync(path.join(HERE, "lowctr-today.html"), "utf8");
const images = JSON.parse(readFileSync(path.join(HERE, "images.json"), "utf8"));
const today = board(copy, { name: "Main", title: "Shown but not clicked today", height: 2300, kit: KIT, images });

/** The screen's own area: the div that opens with its page header, to its matching close. */
function screenArea(html) {
  const start = html.indexOf('<div class="flex flex-col gap-6"><div data-part="page-header"');
  if (start < 0) throw new Error("no screen area");
  let depth = 0;
  const tag = /<(\/?)div\b[^>]*>/g;
  tag.lastIndex = start;
  for (let match = tag.exec(html); match; match = tag.exec(html)) {
    depth += match[1] ? -1 : 1;
    if (depth === 0) return [start, match.index + match[0].length];
  }
  throw new Error("unclosed screen area");
}
const [from, to] = screenArea(today);

const browser = await chromium.launch();
const heights = {};
for (const [way, name] of [["Main", "Main"], ["W1", "Way1"], ["W2", "Way2"], ["W3", "Way3"]]) {
  let html = today;
  if (way !== "Main") {
    const frame = `${today.slice(0, from)}<div id="hk-root"></div><script>window.__WAY=${JSON.stringify(way)}</script><script src="board.js"></script>${today.slice(to)}`
      .replace(`href="${KIT}"`, 'href="kit.css"');
    writeFileSync(path.join(HERE, `frame-${way}.html`), frame);
    const page = await browser.newPage({ viewport: { width: 1440, height: 1000 } });
    const problems = [];
    page.on("pageerror", (error) => problems.push(error.message));
    await page.goto(pathToFileURL(path.join(HERE, `frame-${way}.html`)).href);
    await page.waitForTimeout(2500);
    if (problems.length) console.log(way, "errors:", problems.slice(0, 3));
    // The website has Classifications, so the real chip says so; the kit's stand-in for the server has none.
    const drawn = (await page.evaluate(() => document.getElementById("hk-root").innerHTML))
      .replaceAll('<span aria-hidden="true" class="min-w-0 truncate">Page type</span>', '<span aria-hidden="true" class="min-w-0 truncate">Classification</span>');
    await page.close();
    html = `${today.slice(0, from)}${drawn}${today.slice(to)}`.replace(/<title>[^<]*<\/title>/, `<title>Shown but not clicked, ${name.replace("Way", "way ")}</title>`)
      .replace("<!-- copied-from-app: today's words, so check:drawing doesn't judge their reading age -->\n", "");
  }
  // The board's height, measured at 1440 with the kit.
  const measure = await browser.newPage({ viewport: { width: 1440, height: 1000 } });
  const local = path.join(HERE, `measure-${name}.html`);
  writeFileSync(local, html.replace(`href="${KIT}"`, 'href="kit.css"').replace('<script src="./support.js"></script>', ""));
  await measure.goto(pathToFileURL(local).href);
  await measure.waitForTimeout(800);
  heights[name] = await measure.evaluate(() => Math.ceil(document.documentElement.scrollHeight));
  await measure.screenshot({ path: path.join(HERE, `${name}.png`), fullPage: true });
  await measure.close();
  html = html.replace(/"\$preview":\{"width":1440,"height":\d+\}/, `"$preview":{"width":1440,"height":${heights[name]}}`);
  writeFileSync(path.join(OUT, `${name}.dc.html`), html);
}
await browser.close();
writeFileSync(path.join(HERE, "heights.json"), JSON.stringify(heights));
console.log(heights);
