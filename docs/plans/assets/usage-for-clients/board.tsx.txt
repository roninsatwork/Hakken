type Booked = { host: string; kinds: string[]; nextAt: number; dec: number };
const KIND_NAME: Record<string, string> = { rankings: "Rankings", aiAnswers: "AI answers", backlinks: "Backlinks", siteAudit: "Site audit" };
const checks = (COMING as unknown as { checks: { kind: string; nextAt: number; nextMonth: number; website: { host: string } | null }[] }).checks;
const BOOKED: Booked[] = Object.values(checks.reduce<Record<string, Booked>>((all, check) => {
  const host = check.website?.host ?? "";
  const row = all[host] ?? (all[host] = { host, kinds: [], nextAt: check.nextAt, dec: 0 });
  row.kinds.push(check.kind); row.dec += check.nextMonth; row.nextAt = Math.min(row.nextAt, check.nextAt);
  return all;
}, {})).sort((a, b) => b.dec - a.dec || a.host.localeCompare(b.host));
const dayMonth = (at: number) => new Date(at).toLocaleDateString("en-GB", { day: "numeric", month: "long", timeZone: "Europe/London" });

        sort={{ key: "credits", direction: "desc", onSort: noop }}
        empty={{ label: "Nothing is booked yet." }}

        columns={[
          { key: "name", header: "Website", sortable: true, className: `${CUT_COLUMN.first} min-w-[160px]`, cell: (row) => <SiteName host={row.host} /> },
          { key: "checks", header: "What runs", className: CUT_COLUMN.second, cell: (row) => <What text={row.kinds.map((kind) => KIND_NAME[kind] ?? kind).sort().join(", ")} /> },

          { key: "credits", header: "Credits", align: "right", sortable: true, cell: (row) => mono(fmt(row.dec), true) },

}
